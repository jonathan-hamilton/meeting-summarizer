### **File Size Validation**
```csharp
public class FastTranscriptionValidator
{
    public const long MAX_FILE_SIZE = 300 * 1024 * 1024; // 300MB
    public const int MAX_DURATION_HOURS = 2;
    
    public static ValidationResult ValidateForFastTranscription(IFormFile file)
    {
        if (file.Length > MAX_FILE_SIZE)
            return ValidationResult.Fail($"File size exceeds 300MB limit");
            
        var estimatedDuration = EstimateDuration(file.Length, file.FileName);
        if (estimatedDuration > 120) // 2 hours
            return ValidationResult.Fail($"Estimated duration exceeds 2-hour limit");
            
        return ValidationResult.Success();
    }
}
```
